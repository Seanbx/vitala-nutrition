# 用户画像与动态学习机制设计

## 一、信息提取容错机制

### 1.1 置信度定义

**置信度分数（0-1）：** 表示系统对提取信息准确性的信心程度

| 置信度范围 | 说明 | 来源示例 |
|------------|------|----------|
| 0.9-1.0 | 高置信度 | 用户明确陈述 |
| 0.7-0.9 | 中置信度 | 用户间接表达 |
| 0.5-0.7 | 低置信度 | 模糊表述、推断 |
| 0.3-0.5 | 极低置信度 | 猜测、不确定 |

### 1.2 信息来源分类

```python
class InformationSource:
    EXPLICIT_STATEMENT = "user_stated"  # 用户明确陈述
    INDIRECT_EXPRESSION = "user_implied"  # 用户间接表达
    BEHAVIOR_INFERENCE = "behavior_inferred"  # 行为推断
    SYSTEM_ESTIMATE = "system_estimated"  # 系统估算
```

### 1.3 信息提取规则

| 用户表述 | 提取信息 | 置信度 | 来源 |
|----------|----------|--------|------|
| "我要减肥" | fitness_goal=fat_loss | 0.95 | explicit |
| "我不太爱吃甜的" | taste_preference.low_sweet | 0.7 | implied |
| "我最近在健身" | activity_level=moderate | 0.6 | implied |
| 用户连续选择低卡菜品 | taste_preference.low_calorie | 0.7 | behavior |
| "奶茶可以接受" | taste_preference.sweet (例外) | 0.5 | implied |

### 1.4 矛盾信息处理

**检测规则：**
```python
class ContradictionDetector:
    def detect_contradiction(self, old_info, new_info):
        """检测新旧信息是否矛盾"""
        if old_info["field"] != new_info["field"]:
            return False
        
        if old_info["value"] == new_info["value"]:
            return False
        
        # 计算置信度差异
        confidence_diff = abs(old_info["confidence"] - new_info["confidence"])
        
        if confidence_diff < 0.2:
            # 置信度相近，需要确认
            return {
                "contradicted": True,
                "action": "ask_user",
                "message": f"你之前说{old_info['value']}，现在说{new_info['value']}，哪个是正确的？"
            }
        elif new_info["confidence"] > old_info["confidence"]:
            # 新信息置信度更高，覆盖旧信息
            return {
                "contradicted": True,
                "action": "override",
                "message": "已更新你的信息"
            }
        else:
            # 旧信息置信度更高，保留旧信息
            return {
                "contradicted": True,
                "action": "keep_old",
                "message": "保留之前的信息"
            }
```

### 1.5 信息更新策略

```python
class UserProfileUpdater:
    def update_field(self, user_id, field, new_value, confidence, source):
        """更新用户画像字段"""
        old_info = self.get_field(user_id, field)
        
        # 检查是否矛盾
        if old_info:
            contradiction = self.detect_contradiction(old_info, {
                "field": field,
                "value": new_value,
                "confidence": confidence
            })
            
            if contradiction["contradicted"]:
                if contradiction["action"] == "ask_user":
                    return self.ask_user_for_clarification(field, old_info, new_value)
                elif contradiction["action"] == "keep_old":
                    return {"updated": False, "reason": "kept_old"}
        
        # 更新字段
        self.set_field(user_id, field, {
            "value": new_value,
            "confidence": confidence,
            "source": source,
            "timestamp": datetime.now(),
            "previous_values": self.get_previous_values(user_id, field)
        })
        
        return {"updated": True}
```

### 2.2 缺失信息提示

```python
class MissingInfoPrompter:
    def generate_prompt(self, user_profile):
        """生成缺失信息提示"""
        missing_required = []
        missing_recommended = []
        
        for field, config in FieldClassification.FIELD_TYPES.items():
            if field not in user_profile or user_profile[field].get("is_missing"):
                if config["type"] == FieldClassification.REQUIRED:
                    missing_required.append(field)
                elif config["type"] == FieldClassification.RECOMMENDED:
                    missing_recommended.append(field)
        
        prompt = ""
        
        if missing_required:
            prompt += "⚠️ 为了给你安全的饮食建议，请先补充以下信息：\n\n"
            for field in missing_required:
                prompt += f"- {self.get_field_description(field)}\n"
        
        if missing_recommended:
            prompt += "\n📝 补充以下信息可以提升推荐准确性：\n\n"
            for field in missing_recommended:
                prompt += f"- {self.get_field_description(field)}\n"
        
        return prompt
    
    def get_field_description(self, field):
        """获取字段描述"""
        descriptions = {
            "gender": "你的性别",
            "age": "你的年龄",
            "height": "你的身高（cm）",
            "weight": "你的体重（kg）",
            "fitness_goal": "你的健身目标（减脂/增肌/维持）",
            "medical_conditions": "你是否有糖尿病、高血压等慢性疾病？",
            "food_allergies": "你是否有食物过敏？",
            "activity_level": "你的日常活动水平（久坐/轻度/中度/高强度）",
            "taste_preferences": "你喜欢什么口味？",
        }
        return descriptions.get(field, field)
```

### 3.3 黑名单管理

```python
class BlacklistManager:
    def __init__(self):
        self.blacklist = {}  # user_id -> {field: value, count, last_updated}
    
    def add_to_blacklist(self, user_id, field, value, is_permanent=False):
        """添加到黑名单"""
        if user_id not in self.blacklist:
            self.blacklist[user_id] = {}
        
        key = f"{field}:{value}"
        
        if key in self.blacklist[user_id]:
            # 已存在，增加计数
            self.blacklist[user_id][key]["count"] += 1
            self.blacklist[user_id][key]["last_updated"] = datetime.now()
        else:
            # 新增
            self.blacklist[user_id][key] = {
                "field": field,
                "value": value,
                "count": 1,
                "is_permanent": is_permanent,
                "created_at": datetime.now(),
                "last_updated": datetime.now()
            }
        
        # 检查是否达到永久标记阈值
        if not is_permanent and self.blacklist[user_id][key]["count"] >= 3:
            self.blacklist[user_id][key]["is_permanent"] = True
    
    def is_blacklisted(self, user_id, field, value):
        """检查是否在黑名单中"""
        if user_id not in self.blacklist:
            return False
        
        key = f"{field}:{value}"
        return self.blacklist[user_id].get(key, {}).get("is_permanent", False)
```

```python
class FeedbackProcessor:
    def __init__(self):
        self.blacklist_manager = BlacklistManager()
    
    def process_feedback(self, user_id, recipe_id, action, feedback_tags):
        """处理用户反馈"""
        # 1. 记录反馈
        feedback = FeedbackRecord(recipe_id, action, feedback_tags)
        self.save_feedback(user_id, feedback)
        
        # 2. 如果是拒绝，分析原因
        if action == "rejected":
            self.process_rejection(user_id, recipe_id, feedback_tags)
        
        # 3. 更新用户偏好
        self.update_user_preferences(user_id, action, feedback_tags)
    
    def process_rejection(self, user_id, recipe_id, feedback_tags):
        """处理拒绝反馈"""
        # 获取菜谱信息
        recipe = self.get_recipe(recipe_id)
        
        # 分析拒绝原因
        for tag in feedback_tags:
            if tag in [FeedbackTags.TOO_SPICY, FeedbackTags.TOO_SWEET, FeedbackTags.TOO_SALTY]:
                # 口味问题，更新口味偏好
                self.update_taste_preference(user_id, tag)
            
            elif tag == FeedbackTags.TOO_HIGH_CALORIE:
                # 热量问题，降低热量目标
                self.adjust_calorie_target(user_id, -100)
            
            elif tag in [FeedbackTags.NOT_IN_MOOD, FeedbackTags.ATE_ALREADY]:
                # 临时原因，不加入黑名单
                pass
            
            elif tag == FeedbackTags.NOT_TASTY:
                # 不好吃，检查是否累计3次
                self.blacklist_manager.add_to_blacklist(
                    user_id, "recipe_id", recipe_id
                )
```

## 六、实现代码

### 6.1 信息提取器

```python
class InformationExtractor:
    def __init__(self):
        self.patterns = {
            "gender": {
                "male": ["男", "男生", "男性", "man"],
                "female": ["女", "女生", "女性", "woman"]
            },
            "fitness_goal": {
                "fat_loss": ["减肥", "减脂", "瘦身", "瘦"],
                "muscle_gain": ["增肌", "增重", "壮"],
                "maintain": ["维持", "保持"]
            }
        }
    
    def extract_from_text(self, text, existing_profile=None):
        """从文本中提取信息"""
        extracted = []
        
        for field, patterns in self.patterns.items():
            for value, keywords in patterns.items():
                for keyword in keywords:
                    if keyword in text:
                        confidence = self.calculate_confidence(text, keyword)
                        extracted.append({
                            "field": field,
                            "value": value,
                            "confidence": confidence,
                            "source": "user_stated"
                        })
        
        return extracted
    
    def calculate_confidence(self, text, keyword):
        """计算置信度"""
        # 简单规则：明确陈述置信度高
        if f"我是{keyword}" in text or f"我{keyword}" in text:
            return 0.95
        elif keyword in text:
            return 0.8
        return 0.5
```

### 6.2 完整的用户画像管理器

```python
class UserProfileManager:
    def __init__(self):
        self.profiles = {}
        self.extractor = InformationExtractor()
        self.updater = UserProfileUpdater()

    def process_message(self, user_id, message):
```
