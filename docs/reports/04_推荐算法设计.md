## 一、特殊人群营养计算

### 1.1 BMR/TDEE计算公式

**标准公式（Mifflin-St Jeor）：**
```python
def calculate_bmr(gender, age, weight, height):
    """计算基础代谢率"""
    if gender == "male":
        bmr = (10 * weight) + (6.25 * height) - (5 * age) + 5
    else:
        bmr = (10 * weight) + (6.25 * height) - (5 * age) - 161
    return bmr

def calculate_tdee(bmr, activity_level):
    """计算每日总消耗"""
    activity_coefficients = {
        "sedentary": 1.2,        # 久坐
        "lightly_active": 1.375, # 轻度活动
        "moderately_active": 1.462,  # 中度活动
        "very_active": 1.55,     # 高强度
        "extremely_active": 1.725   # 极高强度
    }
    return bmr * activity_coefficients.get(activity_level, 1.2)
```

### 1.2 特殊人群修正

```python
class SpecialPopulationCalculator:
    """特殊人群营养计算器"""
    
    def calculate_calorie_target(self, user_profile):
        """计算热量目标"""
        # 基础计算
        bmr = calculate_bmr(
            user_profile['gender'],
            user_profile['age'],
            user_profile['weight'],
            user_profile['height']
        )
        tdee = calculate_tdee(bmr, user_profile['activity_level'])
        
        # 特殊人群修正
        adjustment = self.get_special_adjustment(user_profile)
        adjusted_tdee = tdee * adjustment['calorie_multiplier']
        
        # 目标调整
        goal = user_profile['fitness_goal']
        if goal == "fat_loss":
            target = adjusted_tdee * 0.85  # 15%热量缺口
        elif goal == "muscle_gain":
            target = adjusted_tdee * 1.1   # 10%热量盈余
        else:
            target = adjusted_tdee
        
        # 安全校验
        target = self.apply_safety_limits(target, user_profile)
        
        return {
            "bmr": bmr,
            "tdee": tdee,
            "adjusted_tdee": adjusted_tdee,
            "target_calories": target,
            "adjustment_info": adjustment
        }
    
    def get_special_adjustment(self, user_profile):
        """获取特殊人群调整"""
        adjustment = {
            "calorie_multiplier": 1.0,
            "protein_multiplier": 1.0,
            "carb_limit": None,
            "sodium_limit": None,
            "notes": []
        }
        
        # 孕妇/哺乳期
        if user_profile.get('pregnancy_status') == "pregnant":
            adjustment["calorie_multiplier"] = 1.15
            adjustment["notes"].append("孕期额外需要+300-500kcal")
        
        elif user_profile.get('pregnancy_status') == "breastfeeding":
            adjustment["calorie_multiplier"] = 1.2
            adjustment["notes"].append("哺乳期额外需要+500kcal")
        
        # 老年人（>65岁）
        if user_profile.get('age', 0) > 65:
            adjustment["calorie_multiplier"] = 0.9
            adjustment["notes"].append("老年人基础代谢下降")
        
        # 糖尿病患者
        if "diabetes" in user_profile.get('medical_conditions', []):
            adjustment["carb_limit"] = 0.45  # 碳水不超过45%
            adjustment["notes"].append("糖尿病患者需控制碳水摄入")
        
        # 肾病患者
        if "kidney_disease" in user_profile.get('medical_conditions', []):
            adjustment["protein_multiplier"] = 0.7
            adjustment["notes"].append("肾病患者需限制蛋白质")
        
        # 高血压患者
        if "hypertension" in user_profile.get('medical_conditions', []):
            adjustment["sodium_limit"] = 2000  # 钠不超过2000mg
            adjustment["notes'].append("高血压患者需限制钠摄入")
        
        return adjustment
    
    def apply_safety_limits(self, target_calories, user_profile):
        """应用安全限制"""
        gender = user_profile['gender']
        weight = user_profile['weight']
        
        # 最低热量限制
        min_calories = 1200 if gender == "female" else 1500
        min_from_bmr = weight * 24 * 0.8
        
        safe_minimum = max(min_calories, min_from_bmr)
        
        if target_calories < safe_minimum:
            return safe_minimum
        
        return target_calories
```

```python
class NutrientAllocator:
    """营养素分配器"""
    
    def allocate_nutrients(self, calories_target, user_profile):
        """分配营养素"""
        goal = user_profile['fitness_goal']
        weight = user_profile['weight']
        
        # 蛋白质分配
        protein_per_kg = self.get_protein_recommendation(goal, user_profile)
        protein_grams = weight * protein_per_kg
        protein_calories = protein_grams * 4
        
        # 脂肪分配
        fat_per_kg = 0.8 if goal == "fat_loss" else 1.0
        fat_grams = weight * fat_per_kg
        fat_calories = fat_grams * 9
        
        # 碳水分配（剩余热量）
        remaining_calories = calories_target - protein_calories - fat_calories
        carbs_grams = remaining_calories / 4 if remaining_calories > 0 else 0
        
        # 特殊人群碳水限制
        carb_limit = user_profile.get('carb_limit')
        if carb_limit:
            max_carbs = calories_target * carb_limit / 4
            carbs_grams = min(carbs_grams, max_carbs)
        
        return {
            "protein": {
                "grams": protein_grams,
                "calories": protein_calories,
                "per_kg": protein_per_kg
            },
            "fat": {
                "grams": fat_grams,
                "calories": fat_calories,
                "per_kg": fat_per_kg
            },
            "carbs": {
                "grams": carbs_grams,
                "calories": carbs_grams * 4
            }
        }

        """获取蛋白质推荐量"""

            "fat_loss": 1.8,
            "muscle_gain": 2.0,
            "maintain": 1.5

        # 特殊人群调整
        if "kidney_disease" in user_profile.get('medical_conditions', []):
            return 0.7  # 肾病患者限制蛋白质
        
        return base_recommendation
```

## 二、约束冲突降级策略

### 2.1 约束优先级定义

```python
class ConstraintPriority:
    """约束优先级"""
    
    CRITICAL = 1    # 关键约束（过敏、慢病）
    HIGH = 2        # 高优先级（目标、热量）
    MEDIUM = 3      # 中优先级（口味、偏好）
    LOW = 4         # 低优先级（其他）
    
    # 约束类型定义
    CONSTRAINT_TYPES = {
        "allergen_exclude": CRITICAL,
        "medical_condition": CRITICAL,
        "calorie_range": HIGH,
        "fitness_goal": HIGH,
        "taste_preference": MEDIUM,
        "cooking_difficulty": LOW,
        "cost_level": LOW,
        "cook_time": LOW
    }
```

### 2.2 约束降级处理器

```python
class ConstraintResolver:
    """约束冲突处理器"""
    
    def __init__(self):
        self.priority = ConstraintPriority()
    
    def resolve_constraints(self, constraints, candidates):
        """解决约束冲突"""
        # 按优先级排序约束
        sorted_constraints = self.sort_by_priority(constraints)
        
        # 尝试满足所有约束
        results = self.apply_all_constraints(candidates, sorted_constraints)
        
        if results:
            return {
                "recipes": results,
                "message": None,
                "constraints_applied": list(constraints.keys())
            }
        
        # 逐步放宽约束
        for constraint_name in sorted_constraints:
            if self.is_optional_constraint(constraint_name):
                results = self.apply_constraints_except(
                    candidates, sorted_constraints, constraint_name
                )
                
                if results:
                    return {
                        "recipes": results,
                        "message": f"没有完全匹配，已放宽「{self.get_constraint_name(constraint_name)}」条件",
                        "constraints_applied": [c for c in sorted_constraints if c != constraint_name]
                    }
        
        # 返回最接近的结果
        return {
            "recipes": candidates[:5],
            "message": "没有找到完全匹配的菜品，为你推荐以下最接近的选择",
            "constraints_applied": []
        }
    
    def sort_by_priority(self, constraints):
        """按优先级排序约束"""
        return sorted(
            constraints.keys(),
            key=lambda x: self.priority.CONSTRAINT_TYPES.get(x, 99)
        )
    
    def is_optional_constraint(self, constraint_name):
        """判断是否为可选约束"""
        priority = self.priority.CONSTRAINT_TYPES.get(constraint_name, 99)
        return priority >= self.priority.MEDIUM
    
    def apply_all_constraints(self, candidates, constraints):
        """应用所有约束"""
        results = candidates.copy()
        
        for constraint_name in constraints:
            results = self.apply_constraint(results, constraint_name)
            if not results:
                return []
        
        return results
    
    def apply_constraints_except(self, candidates, constraints, except_name):
        """应用除指定约束外的所有约束"""
        results = candidates.copy()
        
        for constraint_name in constraints:
            if constraint_name != except_name:
                results = self.apply_constraint(results, constraint_name)
                if not results:
                    return []
        
        return results
    
    def apply_constraint(self, recipes, constraint_name):
        """应用单个约束"""
        # 这里根据具体约束类型进行过滤
        # 实际实现需要根据约束类型调用不同的过滤逻辑
        return recipes
    
    def get_constraint_name(self, constraint_name):
        """获取约束的中文名称"""
        names = {
            "allergen_exclude": "过敏原排除",
            "calorie_range": "热量范围",
            "taste_preference": "口味偏好",
            "cooking_difficulty": "烹饪难度",
            "cost_level": "成本预算",
            "cook_time": "烹饪时间"
        }
        return names.get(constraint_name, constraint_name)
```

```python
class WeeklyPlanPrinciples:
```

### 3.2 周计划生成器

```python
class WeeklyPlanGenerator:
    """周计划生成器"""
    
    def __init__(self, knowledge_base):
        self.kb = knowledge_base
        self.principles = WeeklyPlanPrinciples()
    
    def generate_weekly_plan(self, user_profile):
        """生成周计划"""
        # 1. 计算周营养目标
        weekly_targets = self.calculate_weekly_targets(user_profile)
        
        # 2. 生成每日计划
        daily_plans = []
        used_recipes = []
        used_ingredients = []
        
        for day in range(7):
            daily_plan = self.generate_daily_plan(
                user_profile,
                weekly_targets,
                daily_plans,  # 之前的计划
                used_recipes,
                used_ingredients
            )
            
            daily_plans.append(daily_plan)
            used_recipes.extend([meal['recipe_id'] for meal in daily_plan['meals']])
            used_ingredients.extend(daily_plan['ingredients'])
        
        # 3. 检查并调整
        daily_plans = self.adjust_plan(daily_plans, weekly_targets)
        
        return {
            "weekly_targets": weekly_targets,
            "daily_plans": daily_plans,
            "shopping_list": self.generate_shopping_list(daily_plans)
        }
    
    def calculate_weekly_targets(self, user_profile):
        """计算周营养目标"""
        calculator = SpecialPopulationCalculator()
        daily_target = calculator.calculate_calorie_target(user_profile)
        
        allocator = NutrientAllocator()
        daily_nutrients = allocator.allocate_nutrients(
            daily_target['target_calories'],
            user_profile
        )
        
        return {
            "daily_calories": daily_target['target_calories'],
            "daily_protein": daily_nutrients['protein']['grams'],
            "daily_fat": daily_nutrients['fat']['grams'],
            "daily_carbs": daily_nutrients['carbs']['grams'],
            "weekly_calories": daily_target['target_calories'] * 7,
            "weekly_protein": daily_nutrients['protein']['grams'] * 7,
            "weekly_fat": daily_nutrients['fat']['grams'] * 7,
            "weekly_carbs": daily_nutrients['carbs']['grams'] * 7
        }
    
    def generate_daily_plan(self, user_profile, weekly_targets, previous_days, used_recipes, used_ingredients):
        """生成每日计划"""
        # 计算今日目标
        daily_calories = weekly_targets['daily_calories']
        daily_protein = weekly_targets['daily_protein']
        
        # 三餐热量分配
        meal_distribution = {
            "breakfast": 0.25,  # 早餐25%
            "lunch": 0.40,      # 午餐40%
            "dinner": 0.30,     # 晚餐30%
            "snack": 0.05       # 加餐5%
        }
        
        meals = []
        
        # 生成早餐
        breakfast = self.generate_meal(
            "breakfast",
            daily_calories * meal_distribution['breakfast'],
            daily_protein * meal_distribution['breakfast'],
            user_profile,
            used_recipes
        )
        meals.append(breakfast)
        
        # 生成午餐
        lunch = self.generate_meal(
            "lunch",
            daily_calories * meal_distribution['lunch'],
            daily_protein * meal_distribution['lunch'],
            user_profile,
            used_recipes
        )
        meals.append(lunch)
        
        # 生成晚餐
        dinner = self.generate_meal(
            "dinner",
            daily_calories * meal_distribution['dinner'],
            daily_protein * meal_distribution['dinner'],
            user_profile,
            used_recipes
        )
        meals.append(dinner)
        
        # 生成加餐（可选）
        if daily_calories * meal_distribution['snack'] > 100:
            snack = self.generate_meal(
                "snack",
                daily_calories * meal_distribution['snack'],
                daily_protein * meal_distribution['snack'],
                user_profile,
                used_recipes
            )
            meals.append(snack)
        
        # 计算今日实际营养
        total_nutrients = self.calculate_total_nutrients(meals)
        
        return {
            "date": self.get_next_date(len(previous_days)),
            "meals": meals,
            "total_nutrients": total_nutrients,
            "ingredients": self.extract_ingredients(meals)
        }
    
    def generate_meal(self, meal_type, target_calories, target_protein, user_profile, used_recipes):
        """生成单餐"""
        # 搜索候选菜谱
        candidates = self.kb.search_recipes(
            query=f"{meal_type} 减脂",
            user_id=user_profile['user_id'],
            top_k=10
        )
        
        # 过滤已使用的菜谱
        candidates = [r for r in candidates if r['id'] not in used_recipes]
        
        # 过滤不合适的菜谱
        candidates = self.filter_meal_candidates(candidates, meal_type, target_calories)
        
        # 选择最佳菜谱
        if candidates:
            selected = self.select_best_recipe(candidates, target_calories, target_protein)
        else:
            # 无候选，使用默认推荐
            selected = self.get_default_meal(meal_type)
        
        return selected
    
    def filter_meal_candidates(self, candidates, meal_type, target_calories):
        """过滤候选菜谱"""
        filtered = []
        
        for recipe in candidates:
            # 检查餐次是否匹配
            if recipe.get('meal_type') != meal_type:
                continue
            
            # 检查热量范围（允许±30%波动）
            calorie_diff = abs(recipe['calories'] - target_calories) / target_calories
            if calorie_diff > 0.3:
                continue
            
            # 检查过敏原
            if not self.is_recipe_safe(recipe, user_profile):
                continue
            
            filtered.append(recipe)
        
        return filtered
    
    def select_best_recipe(self, candidates, target_calories, target_protein):
        """选择最佳菜谱"""
        scored = []
        
        for recipe in candidates:
            score = self.calculate_recipe_score(recipe, target_calories, target_protein)
            scored.append((recipe, score))
        
        scored.sort(key=lambda x: x[1], reverse=True)
        return scored[0][0] if scored else None
    
    def calculate_recipe_score(self, recipe, target_calories, target_protein):
        """计算菜谱分数"""
        score = 0
        
        # 热量匹配度
        calorie_diff = abs(recipe['calories'] - target_calories)
        score += max(0, 100 - calorie_diff)
        
        # 蛋白质匹配度
        protein_diff = abs(recipe.get('protein', 0) - target_protein)
        score += max(0, 50 - protein_diff)
        
        # 难度适中
        if recipe.get('difficulty', 3) <= 3:
            score += 20
        
        # 烹饪时间适中
        if recipe.get('cook_time', 30) <= 30:
            score += 10
        
        return score
    
    def adjust_plan(self, daily_plans, weekly_targets):
        """调整周计划"""
        # 检查周营养达标情况
        weekly_nutrients = self.calculate_weekly_nutrients(daily_plans)
        
        # 如果周营养不达标，调整最后几天
        if abs(weekly_nutrients['calories'] - weekly_targets['weekly_calories']) > weekly_targets['weekly_calories'] * 0.1:
            daily_plans = self.adjust_last_days(daily_plans, weekly_targets)
        
        return daily_plans
    
    def generate_shopping_list(self, daily_plans):
        """生成购物清单"""
        ingredients = {}
        
        for day in daily_plans:
            for meal in day['meals']:
                for ingredient in meal.get('ingredients', []):
                    name = ingredient['name']
                    amount = ingredient['amount']
                    
                    if name in ingredients:
                        ingredients[name] += amount
                    else:
                        ingredients[name] = amount
        
        # 转换为列表格式
        shopping_list = [
            {"name": name, "total_amount": amount}
            for name, amount in ingredients.items()
        ]
        
        return shopping_list
```

## 四、推荐评分系统

### 4.1 评分维度

```python
class RecipeScorer:
    """菜谱评分器"""
    
    def score_recipe(self, recipe, user_context):
        """计算菜谱总分"""
        scores = {
            "nutrition_score": self.score_nutrition(recipe, user_context),
            "preference_score": self.score_preference(recipe, user_context),
            "safety_score": self.score_safety(recipe, user_context),
            "practicality_score": self.score_practicality(recipe, user_context),
            "diversity_score": self.score_diversity(recipe, user_context)
        }
        
        # 加权总分
        weights = {
            "nutrition_score": 0.3,
            "preference_score": 0.25,
            "safety_score": 0.25,
            "practicality_score": 0.1,
            "diversity_score": 0.1
        }
        
        total_score = sum(scores[k] * weights[k] for k in scores)
        
        return {
            "total_score": total_score,
            "scores": scores,
            "breakdown": self.get_score_breakdown(scores, weights)
        }
    
    def score_nutrition(self, recipe, user_context):
        """营养评分"""
        score = 0
        target_calories = user_context['calorie_target']
        
        # 热量匹配度
        calorie_diff = abs(recipe['calories'] - target_calories * 0.35) / (target_calories * 0.35)
        score += max(0, 100 - calorie_diff * 100)
        
        # 蛋白质含量
        if recipe.get('protein', 0) > 20:
            score += 30
        elif recipe.get('protein', 0) > 15:
            score += 20
        
        # 脂肪含量（减脂期低脂加分）
        if user_context.get('fitness_goal') == "fat_loss":
            if recipe.get('fat', 0) < 10:
                score += 20
        
        return min(100, score)
    
    def score_preference(self, recipe, user_context):
        """偏好评分"""
        score = 50  # 基础分
        
        # 口味匹配
        taste_prefs = user_context.get('taste_preferences', {})
        if taste_prefs:
            # 根据口味偏好评分
            score += 20
        
        # 历史接受记录
        if recipe['id'] in user_context.get('accepted_recipes', []):
            score += 30
        
        # 历史拒绝记录
        if recipe['id'] in user_context.get('rejected_recipes', []):
            score -= 30
        
        return max(0, min(100, score))
    
    def score_safety(self, recipe, user_context):
        """安全评分"""
        score = 100
        
        # 过敏原检查
        allergens = recipe.get('allergens', [])
        user_allergens = user_context.get('allergies', [])
        
        if set(allergens) & set(user_allergens):
            score = 0  # 含有过敏原，直接0分
        
        # 过敏原信息缺失
        if allergens is None:
            if user_allergens:
                score -= 30  # 用户有过敏史，但菜谱过敏原信息缺失
        
        # 慢性疾病检查
        medical_conditions = user_context.get('medical_conditions', [])
        if "diabetes" in medical_conditions:
            if recipe.get('carbs', 0) > 40:
                score -= 20  # 糖尿病患者，高碳水扣分
        
        return max(0, min(100, score))
    
    def score_practicality(self, recipe, user_context):
        """实用性评分"""
        score = 50
        
        # 烹饪时间
        cook_time = recipe.get('cook_time', 30)
        if cook_time <= 15:
            score += 30
        elif cook_time <= 30:
            score += 20
        elif cook_time <= 45:
            score += 10
        
        # 难度
        difficulty = recipe.get('difficulty', 3)
        if difficulty <= 2:
            score += 20
        elif difficulty <= 3:
            score += 10
        
        # 食材易得性
        availability = recipe.get('ingredient_availability', 'common')
        if availability == 'common':
            score += 10
        
        return min(100, score)
    
    def score_diversity(self, recipe, user_context):
        """多样性评分"""
        score = 50
        
        # 与最近推荐的菜谱是否重复
        recent_recipes = user_context.get('recent_recipes', [])
        if recipe['id'] not in [r['id'] for r in recent_recipes]:
            score += 30
        
        # 食材是否重复
        recipe_ingredients = set(recipe.get('ingredients', []))
        recent_ingredients = set(user_context.get('recent_ingredients', []))
        
        if not recipe_ingredients & recent_ingredients:
            score += 20
        
        return min(100, score)
```

## 五、完整推荐流程

```python
class RecommendationEngine:
    """推荐引擎"""
    
    def __init__(self, knowledge_base):
        self.kb = knowledge_base
        self.safety_validator = SafetyValidator()
        self.allergen_validator = AllergenValidator()
        self.constraint_resolver = ConstraintResolver()
        self.recipe_scorer = RecipeScorer()
    
    def recommend(self, user_id, query, top_k=5):
        """推荐菜谱"""
        # 1. 获取用户画像
        user_profile = self.get_user_profile(user_id)
        
        # 2. 安全校验
        safety_check = self.safety_validator.validate_input(query, user_profile)
        if not safety_check['safe']:
            return {
                "recipes": [],
                "message": safety_check['message']
            }
        
        # 3. 构建约束条件
        constraints = self.build_constraints(user_profile, query)
        
        # 4. 检索候选菜谱
        candidates = self.kb.search_recipes(query, user_id, top_k=top_k * 4)
        
        # 5. 解决约束冲突
        resolved = self.constraint_resolver.resolve_constraints(constraints, candidates)
        
        # 6. 评分排序
        scored = []
        for recipe in resolved['recipes']:
            score = self.recipe_scorer.score_recipe(recipe, {
                "calorie_target": user_profile['nutrition_targets']['calorie_target'],
                "fitness_goal": user_profile['fitness_goal'],
                "allergies": user_profile['food_allergies']['value'],
                "medical_conditions": user_profile['medical_conditions']['value'],
                "taste_preferences": user_profile['preference_info']['taste_preferences'],
                "accepted_recipes": user_profile.get('feedback_summary', {}).get('favorite_recipes', []),
                "rejected_recipes": user_profile.get('feedback_summary', {}).get('disliked_recipes', [])
            })
            scored.append((recipe, score))
        
        # 7. 排序取Top-K
        scored.sort(key=lambda x: x[1]['total_score'], reverse=True)
        top_recipes = [recipe for recipe, score in scored[:top_k]]
        
        # 8. 添加风险提示
        risk_notes = self.generate_risk_notes(user_profile)
        
        return {
            "recipes": top_recipes,
            "message": resolved.get('message'),
            "risk_notes": risk_notes,
            "constraints_applied": resolved.get('constraints_applied')
        }
    
    def build_constraints(self, user_profile, query):
        """构建约束条件"""
        constraints = {
            "allergen_exclude": user_profile['food_allergies']['value'],
            "calorie_range": {
                "min": user_profile['nutrition_targets']['calorie_target'] * 0.2,
                "max": user_profile['nutrition_targets']['calorie_target'] * 0.5
            },
            "fitness_goal": user_profile['fitness_goal']
        }
        
        # 添加口味偏好约束
        if user_profile['preference_info'].get('taste_preferences'):
            constraints['taste_preference'] = user_profile['preference_info']['taste_preferences']
        
        return constraints
    
    def generate_risk_notes(self, user_profile):
        """生成风险提示"""
        notes = []
        
        # 健康信息缺失
        if user_profile.get('medical_conditions', {}).get('is_missing'):
            notes.append("⚠️ 健康信息不完整，推荐仅供参考，请咨询医生")
        
        # 过敏信息缺失
        if user_profile.get('food_allergies', {}).get('is_missing'):
            notes.append("⚠️ 过敏信息不完整，请仔细核实食材")
        
        return notes
```
