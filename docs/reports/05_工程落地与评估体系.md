```python
class MetricsCollector:
    """指标采集器"""
    
    def __init__(self):
        self.metrics_data = {}
    
    def track_recommendation(self, user_id, recipe_id, action):
        """跟踪推荐行为"""
        key = f"recommendation_{user_id}"
        
        if key not in self.metrics_data:
            self.metrics_data[key] = {
                "total": 0,
                "accepted": 0,
                "rejected": 0,
                "clicked": 0
            }
        
        self.metrics_data[key]["total"] += 1
        
        if action == "accepted":
            self.metrics_data[key]["accepted"] += 1
        elif action == "rejected":
            self.metrics_data[key]["rejected"] += 1
        elif action == "clicked":
            self.metrics_data[key]["clicked"] += 1
    
    def track_meal(self, user_id, meal_data):
        """跟踪饮食记录"""
        key = f"meal_{user_id}_{meal_data['date']}"
        
        self.metrics_data[key] = {
            "total_calories": meal_data['total_calories'],
            "total_protein": meal_data['total_protein'],
            "total_fat": meal_data['total_fat'],
            "total_carbs": meal_data['total_carbs'],
            "timestamp": datetime.now()
        }
    
    def track_feedback(self, user_id, recipe_id, rating, comment=None):
        """跟踪反馈"""
        key = f"feedback_{user_id}"
        
        if key not in self.metrics_data:
            self.metrics_data[key] = {
                "ratings": [],
                "comments": []
            }
        
        self.metrics_data[key]["ratings"].append(rating)
        
        if comment:
            self.metrics_data[key]["comments"].append({
                "recipe_id": recipe_id,
                "comment": comment,
                "timestamp": datetime.now()
            })
    
    def calculate_metrics(self, user_id=None):
        """计算指标"""
        if user_id:
            return self.calculate_user_metrics(user_id)
        else:
            return self.calculate_overall_metrics()
    
    def calculate_user_metrics(self, user_id):
        """计算用户指标"""
        metrics = {}
        
        # 推荐接受率
        rec_data = self.metrics_data.get(f"recommendation_{user_id}", {})
        if rec_data.get("total", 0) > 0:
            metrics["acceptance_rate"] = rec_data["accepted"] / rec_data["total"]
            metrics["rejection_rate"] = rec_data["rejected"] / rec_data["total"]
        
        # 热量达标率
        meal_keys = [k for k in self.metrics_data.keys() if k.startswith(f"meal_{user_id}_")]
        if meal_keys:
            total_days = len(meal_keys)
            days_within_range = sum(
                1 for k in meal_keys
                if 1200 <= self.metrics_data[k]["total_calories"] <= 1800
            )
            metrics["calorie_compliance"] = days_within_range / total_days
        
        # 满意度
        feedback_data = self.metrics_data.get(f"feedback_{user_id}", {})
        if feedback_data.get("ratings"):
            metrics["satisfaction_score"] = sum(feedback_data["ratings"]) / len(feedback_data["ratings"])
        
        return metrics
    
    def generate_report(self):
        """生成评估报告"""
        overall_metrics = self.calculate_overall_metrics()
        
        report = {
            "summary": overall_metrics,
            "details": {
                "recommendation": self.get_recommendation_details(),
                "nutrition": self.get_nutrition_details(),
                "satisfaction": self.get_satisfaction_details()
            },
            "recommendations": self.generate_improvement_recommendations(overall_metrics)
        }
        
        return report
    
    def generate_improvement_recommendations(self, metrics):
        """生成改进建议"""
        recommendations = []
        
        # 检查各项指标
        for category, metric_defs in [
            ("recommendation", EvaluationMetrics.RECOMMENDATION_METRICS),
            ("nutrition", EvaluationMetrics.NUTRITION_METRICS),
            ("satisfaction", EvaluationMetrics.SATISFACTION_METRICS)
        ]:
            for metric_name, metric_def in metric_defs.items():
                current_value = metrics.get(metric_name)
                target_value = metric_def["target"]
                
                if current_value is not None and current_value < target_value:
                    recommendations.append({
                        "metric": metric_name,
                        "current": current_value,
                        "target": target_value,
                        "gap": target_value - current_value,
                        "suggestion": self.get_metric_suggestion(metric_name, current_value, target_value)
                    })
        
        return recommendations
    
    def get_metric_suggestion(self, metric_name, current, target):
        """获取指标改进建议"""
        suggestions = {
            "acceptance_rate": "优化推荐算法，增加用户偏好学习",
            "rejection_rate": "改进反馈机制，更精准理解用户需求",
            "calorie_compliance": "优化热量计算，增加饮食提醒",
            "protein_compliance": "增加高蛋白菜谱推荐",
            "satisfaction_score": "提升推荐质量，增加个性化程度",
            "retention_rate": "增加用户粘性功能，如成就系统"
        }
        return suggestions.get(metric_name, "持续优化")
```

### 2.2 更新处理器

```python
class KnowledgeBaseUpdater:
    """知识库更新器"""
    
    def __init__(self, knowledge_base):
        self.kb = knowledge_base
        self.update_log = []
    
    def update(self, update_type, data=None):
        """执行更新"""
        start_time = datetime.now()
        
        try:
            if update_type == UpdateStrategy.INCREMENTAL:
                result = self.incremental_update(data)
            elif update_type == UpdateStrategy.FULL:
                result = self.full_rebuild()
            elif update_type == UpdateStrategy.SCHEDULED:
                result = self.scheduled_update()
            else:
                raise ValueError(f"Unknown update type: {update_type}")
            
            # 记录更新日志
            self.log_update(update_type, result, start_time)
            
            return result
            
        except Exception as e:
            # 记录错误
            self.log_error(update_type, str(e), start_time)
            raise
    
    def incremental_update(self, data):
        """增量更新"""
        added = 0
        updated = 0
        deleted = 0
        
        # 处理新增菜谱
        if "new_recipes" in data:
            for recipe in data["new_recipes"]:
                # 质量校验
                if self.validate_recipe(recipe):
                    # 添加到知识库
                    self.kb.add_recipe(recipe)
                    added += 1
        
        # 处理更新菜谱
        if "updated_recipes" in data:
            for recipe in data["updated_recipes"]:
                # 更新知识库
                self.kb.update_recipe(recipe)
                updated += 1
        
        # 处理删除菜谱
        if "deleted_recipe_ids" in data:
            for recipe_id in data["deleted_recipe_ids"]:
                # 删除知识库
                self.kb.delete_recipe(recipe_id)
                deleted += 1
        
        # 更新向量索引
        self.kb.update_index_incremental()
        
        return {
            "status": "success",
            "added": added,
            "updated": updated,
            "deleted": deleted
        }
    
    def full_rebuild(self):
        """全量重建"""
        # 清空现有索引
        self.kb.clear_index()
        
        # 重新加载所有数据
        recipes = self.kb.load_all_recipes()
        
        # 重新构建向量索引
        self.kb.build_index(recipes)
        
        return {
            "status": "success",
            "total_recipes": len(recipes)
        }
    
    def scheduled_update(self):
        """定时更新"""
        # 检查数据源是否有更新
        sources = self.kb.get_data_sources()
        
        updated_sources = []
        for source in sources:
            if self.check_source_update(source):
                # 获取增量数据
               增量数据 = self.fetch_incremental_data(source)
                
                # 执行增量更新
                self.incremental_update(增量数据)
                
                updated_sources.append(source['name'])
        
        return {
            "status": "success",
            "updated_sources": updated_sources
        }
    
    def validate_recipe(self, recipe):
        """校验菜谱"""
        # 营养值校验
        if recipe.get('calories', 0) < 0 or recipe.get('calories', 0) > 2000:
            return False
        
        if recipe.get('protein', 0) < 0 or recipe.get('protein', 0) > 100:
            return False
        
        # 必填字段校验
        required_fields = ['name', 'ingredients', 'steps', 'calories']
        for field in required_fields:
            if field not in recipe:
                return False
        
        return True
    
    def log_update(self, update_type, result, start_time):
        """记录更新日志"""
        self.update_log.append({
            "timestamp": datetime.now(),
            "update_type": update_type,
            "duration": (datetime.now() - start_time).total_seconds(),
            "result": result
        })
    
    def log_error(self, update_type, error_message, start_time):
        """记录错误日志"""
        self.update_log.append({
            "timestamp": datetime.now(),
            "update_type": update_type,
            "duration": (datetime.now() - start_time).total_seconds(),
            "error": error_message
        })
```

```python
class VersionControl:
    """版本控制"""
    
    def __init__(self):
        self.versions = []
        self.current_version = None
    
    def create_version(self, data, description):
        """创建新版本"""
        version = {
            "version_id": len(self.versions) + 1,
            "timestamp": datetime.now(),
            "description": description,
            "data_hash": self.calculate_hash(data),
            "data_snapshot": data
        }
        
        self.versions.append(version)
        self.current_version = version
        
        return version
    
    def rollback(self, version_id):
        """回滚到指定版本"""
        target_version = None
        for version in self.versions:
            if version['version_id'] == version_id:
                target_version = version
                break
        
        if not target_version:
            raise ValueError(f"Version {version_id} not found")
        
        # 恢复数据
        self.restore_data(target_version['data_snapshot'])
        
        self.current_version = target_version
        
        return {
            "status": "success",
            "restored_version": version_id
        }
    
    def get_version_history(self):
        """获取版本历史"""
        return [
            {
                "version_id": v['version_id'],
                "timestamp": v['timestamp'],
                "description": v['description']
            }
            for v in self.versions
        ]
```

```python
class DataEncryption:
    """数据加密"""
    
    def __init__(self, encryption_key):
        self.encryption_key = encryption_key
        self.cipher = Fernet(encryption_key)
    
    def encrypt(self, data):
        """加密数据"""
        if isinstance(data, dict):
            data_str = json.dumps(data)
        else:
            data_str = str(data)
        
        encrypted = self.cipher.encrypt(data_str.encode())
        return encrypted
    
    def decrypt(self, encrypted_data):
        """解密数据"""
        decrypted = self.cipher.decrypt(encrypted_data)
        return json.loads(decrypted.decode())
    
    def encrypt_sensitive_fields(self, user_profile):
        """加密敏感字段"""
        sensitive_fields = ['medical_conditions', 'food_allergies', 'health_data']
        
        for field in sensitive_fields:
            if field in user_profile:
                user_profile[field] = self.encrypt(user_profile[field])
        
        return user_profile
    
    def decrypt_sensitive_fields(self, user_profile):
        """解密敏感字段"""
        sensitive_fields = ['medical_conditions', 'food_allergies', 'health_data']
        
        for field in sensitive_fields:
            if field in user_profile:
                user_profile[field] = self.decrypt(user_profile[field])
        
        return user_profile
```

```python
class AccessControl:
    """访问控制"""
    
    def __init__(self):
        self.permissions = {}
    
    def grant_permission(self, user_id, resource, permission):
        """授予权限"""
        if user_id not in self.permissions:
            self.permissions[user_id] = {}
        
        if resource not in self.permissions[user_id]:
            self.permissions[user_id][resource] = []
        
        self.permissions[user_id][resource].append(permission)
    
    def check_permission(self, user_id, resource, permission):
        """检查权限"""
        user_permissions = self.permissions.get(user_id, {})
        resource_permissions = user_permissions.get(resource, [])
        
        return permission in resource_permissions
    
    def revoke_permission(self, user_id, resource, permission):
        """撤销权限"""
        if user_id in self.permissions:
            if resource in self.permissions[user_id]:
                if permission in self.permissions[user_id][resource]:
                    self.permissions[user_id][resource].remove(permission)
```

```python
class DataRetentionPolicy:

            "chat_history": {
                "retention_days": 90,
                "auto_delete": True
            },
            "meal_records": {
                "retention_days": 365,
                "auto_delete": True
            },
            "user_profile": {
                "retention_days": None,  # 永久保留
                "auto_delete": False
            },
            "feedback_history": {
                "retention_days": 180,
                "auto_delete": True
            }
        }
    
    def enforce_retention(self):
        """执行保留策略"""
        deleted_counts = {}
        
        for data_type, policy in self.policies.items():
            if policy['auto_delete'] and policy['retention_days']:
                deleted = self.delete_old_data(data_type, policy['retention_days'])
                deleted_counts[data_type] = deleted
        
        return deleted_counts
    
    def delete_old_data(self, data_type, retention_days):
        """删除过期数据"""
        cutoff_date = datetime.now() - timedelta(days=retention_days)
        
        # 查询过期数据
        old_data = self.query_old_data(data_type, cutoff_date)
        
        # 删除
        deleted_count = 0
        for data in old_data:
            self.delete_data(data)
            deleted_count += 1
        
        return deleted_count
```

```python
class SystemMonitor:
    """系统监控"""
    
    def __init__(self):
        self.metrics = {}
        self.alerts = []
    
    def monitor_recommendation_quality(self):
        """监控推荐质量"""
        # 计算实时指标
        realtime_metrics = self.calculate_realtime_metrics()
        
        # 检查是否异常
        for metric_name, value in realtime_metrics.items():
            threshold = self.get_threshold(metric_name)
            
            if value < threshold['min'] or value > threshold['max']:
                self.trigger_alert(metric_name, value, threshold)
        
        return realtime_metrics
    
    def trigger_alert(self, metric_name, value, threshold):
        """触发告警"""
        alert = {
            "timestamp": datetime.now(),
            "metric": metric_name,
            "value": value,
            "threshold": threshold,
            "severity": self.get_alert_severity(metric_name, value, threshold)
        }
        
        self.alerts.append(alert)
        
        # 发送告警通知
        self.send_notification(alert)
        
        return alert
    
    def get_alert_severity(self, metric_name, value, threshold):
        """获取告警严重程度"""
        if value < threshold['min'] * 0.5 or value > threshold['max'] * 1.5:
            return "critical"
        elif value < threshold['min'] * 0.8 or value > threshold['max'] * 1.2:
            return "warning"
        else:
            return "info"
```
